use std::{alloc::{GlobalAlloc,Layout,System},sync::atomic::{AtomicBool,AtomicUsize,Ordering::Relaxed},hint::black_box,time::Instant};
use paseo_desktop::terminal::{TerminalBuffer,GridSize};
struct Count;static ENABLED:AtomicBool=AtomicBool::new(false);static CALLS:AtomicUsize=AtomicUsize::new(0);static BYTES:AtomicUsize=AtomicUsize::new(0);
unsafe impl GlobalAlloc for Count{unsafe fn alloc(&self,l:Layout)->*mut u8{if ENABLED.load(Relaxed){CALLS.fetch_add(1,Relaxed);BYTES.fetch_add(l.size(),Relaxed);}unsafe{System.alloc(l)}}unsafe fn dealloc(&self,p:*mut u8,l:Layout){unsafe{System.dealloc(p,l)}}unsafe fn realloc(&self,p:*mut u8,l:Layout,n:usize)->*mut u8{if ENABLED.load(Relaxed){CALLS.fetch_add(1,Relaxed);BYTES.fetch_add(n,Relaxed);}unsafe{System.realloc(p,l,n)}}}
#[global_allocator] static ALLOC:Count=Count;
fn visit(buffer:&TerminalBuffer)->u64{let screen=buffer.screen();let mut h=0u64;for cell in screen.cells{h=h.wrapping_mul(31).wrapping_add((cell.row*1000+cell.col)as u64);h=h.wrapping_add((cell.foreground^cell.background)as u64);for ch in cell.chars(){h=h.wrapping_mul(31).wrapping_add(ch as u64)}black_box((cell.bold,cell.italic,cell.underline,cell.columns));}h^screen.cursor.map(|(r,c)|(r*1000+c)as u64).unwrap_or(0)}
fn main(){for (name,cols,rows,text) in [("sparse",120,40,"prompt> echo hello".to_string()),("dense",240,70,("0123456789 abcdefghijklmnopqrstuvwxyz ".repeat(1000))), ("unicode_styles",120,40,("\x1b[1;3;4;38;2;105;200;123m中e\u{301}👩\u{200d}💻한글नमस्तेالعربية\x1b[0m | ".repeat(400)))]{let mut buffer=TerminalBuffer::new(GridSize{rows,cols},10_000);buffer.feed(text.as_bytes());for _ in 0..20{black_box(visit(&buffer));}CALLS.store(0,Relaxed);BYTES.store(0,Relaxed);ENABLED.store(true,Relaxed);let start=Instant::now();let mut checksum=0;for _ in 0..1000{checksum=black_box(visit(black_box(&buffer)));}let ns=start.elapsed().as_nanos();ENABLED.store(false,Relaxed);println!("{{\"workload\":\"{name}\",\"iterations\":1000,\"nsPerFrame\":{},\"allocationsPerFrame\":{},\"allocatedBytesPerFrame\":{},\"checksum\":{checksum}}}",ns/1000,CALLS.load(Relaxed)/1000,BYTES.load(Relaxed)/1000);}}
